cmake_minimum_required(VERSION 3.22)
project(green2_cli_tool VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(GREEN2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(GREEN2_HOSTAP_DIR "" CACHE PATH "Compatible hostap PASN source/build")
set(GREEN2_LIBNL_INCLUDE_DIR "" CACHE PATH "Cross-build libnl include directory")
set(GREEN2_LIBNL_LIBRARY "" CACHE FILEPATH "Cross-build combined libnl library")
set(GREEN2_CRYPTO_LIBRARY "" CACHE FILEPATH "Cross-build crypto provider library")
set(GREEN2_PASN_LIBRARY "" CACHE FILEPATH "Compatible prebuilt libpasn.so")
set(GREEN2_ANDROID_ADAPTER_SOURCE "" CACHE FILEPATH "Optional Android HAL defaults adapter")
if(NOT EXISTS "${GREEN2_HOSTAP_DIR}/src/pasn/pasn_common.h")
    message(FATAL_ERROR "Set GREEN2_HOSTAP_DIR to the matching hostap PASN source/build")
endif()
if(NOT GREEN2_PASN_LIBRARY)
    find_library(GREEN2_PASN_LIBRARY_FOUND NAMES pasn PATHS "${GREEN2_HOSTAP_DIR}/wpa_supplicant" NO_DEFAULT_PATH REQUIRED)
    set(GREEN2_PASN_LIBRARY "${GREEN2_PASN_LIBRARY_FOUND}")
endif()
if(GREEN2_LIBNL_LIBRARY)
    if(NOT GREEN2_LIBNL_INCLUDE_DIR)
        message(FATAL_ERROR "Cross-builds require GREEN2_LIBNL_INCLUDE_DIR")
    endif()
    set(NL_LIBRARIES "${GREEN2_LIBNL_LIBRARY}")
    set(NL_INCLUDES "${GREEN2_LIBNL_INCLUDE_DIR}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(NL REQUIRED IMPORTED_TARGET libnl-3.0 libnl-genl-3.0)
    set(NL_LIBRARIES PkgConfig::NL)
endif()
if(NOT GREEN2_CRYPTO_LIBRARY)
    find_package(OpenSSL REQUIRED)
    set(GREEN2_CRYPTO_LIBRARY OpenSSL::Crypto)
endif()
add_library(green2_protocol STATIC "${GREEN2_ROOT}/shared/src/protocol_common.c")
target_include_directories(green2_protocol PUBLIC "${GREEN2_ROOT}/shared/include")
add_library(green2_sha256 STATIC "${GREEN2_ROOT}/shared/src/sha256.c")
target_include_directories(green2_sha256 PUBLIC "${GREEN2_ROOT}/shared/include")
add_library(green2_pasn SHARED src/pasn/pasn_engine.c)
target_include_directories(green2_pasn PUBLIC include PRIVATE "${GREEN2_HOSTAP_DIR}/src" "${GREEN2_HOSTAP_DIR}/src/utils" ${NL_INCLUDES})
target_compile_definitions(green2_pasn PRIVATE CONFIG_SAE CONFIG_PASN CONFIG_NAN _GNU_SOURCE)
target_compile_options(green2_pasn PRIVATE -Wall)
target_link_libraries(green2_pasn PRIVATE green2_protocol "${GREEN2_PASN_LIBRARY}" "${GREEN2_CRYPTO_LIBRARY}" ${NL_LIBRARIES})
if(GREEN2_ANDROID_ADAPTER_SOURCE)
    target_sources(green2_pasn PRIVATE "${GREEN2_ANDROID_ADAPTER_SOURCE}")
    target_link_libraries(green2_pasn PRIVATE log)
endif()
set_target_properties(green2_pasn PROPERTIES OUTPUT_NAME green2pasn BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
foreach(tool green2_cli pasn_live)
    add_executable(${tool} src/main.c src/comm_helper.c)
    target_include_directories(${tool} PRIVATE include ${NL_INCLUDES})
    target_link_libraries(${tool} PRIVATE green2_pasn ${NL_LIBRARIES})
    set_target_properties(${tool} PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endforeach()
add_executable(pasn_vendor_events src/comm_helper.c)
target_compile_definitions(pasn_vendor_events PRIVATE GREEN2_VENDOR_EVENTS_MAIN)
target_include_directories(pasn_vendor_events PRIVATE include ${NL_INCLUDES})
target_link_libraries(pasn_vendor_events PRIVATE ${NL_LIBRARIES})
option(GREEN2_BUILD_NAN_TOOLS "Build Linux publisher/libnanapi" OFF)
if(GREEN2_BUILD_NAN_TOOLS)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
    find_package(Threads REQUIRED)
    set(NAN_API_SOURCES src/nan/nan_event.c src/nan/nan_scenario.c src/nan/rtw_nan_cmd.c)
    add_library(nanapi SHARED ${NAN_API_SOURCES})
    target_include_directories(nanapi PUBLIC include/nan "${GREEN2_ROOT}/shared/include")
    target_link_libraries(nanapi PRIVATE green2_sha256 ${NL_LIBRARIES} Threads::Threads)
    add_executable(rtw_aware src/publisher/rtw_aware.c)
    target_link_libraries(rtw_aware PRIVATE nanapi PkgConfig::JSONC Threads::Threads)
    set_target_properties(rtw_aware PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
    install(TARGETS nanapi rtw_aware LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
    install(DIRECTORY include/nan DESTINATION include)
endif()
include(CTest)
if(BUILD_TESTING)
    foreach(name test_nan_pasn_core test_nan_pasn_setup test_nan_pasn_retransmit)
        add_executable(${name} tests/${name}.c)
        target_compile_definitions(${name} PRIVATE CONFIG_SAE CONFIG_PASN CONFIG_NAN _GNU_SOURCE)
        target_compile_options(${name} PRIVATE -ffunction-sections -fdata-sections)
        target_include_directories(${name} PRIVATE include "${GREEN2_HOSTAP_DIR}/src" "${GREEN2_HOSTAP_DIR}/src/utils" ${NL_INCLUDES})
        target_link_libraries(${name} PRIVATE green2_protocol "${GREEN2_PASN_LIBRARY}" "${GREEN2_CRYPTO_LIBRARY}" ${NL_LIBRARIES})
        target_link_options(${name} PRIVATE -Wl,--gc-sections)
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
endif()
install(TARGETS green2_pasn green2_cli pasn_live pasn_vendor_events LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/green2_pasn.h "${GREEN2_ROOT}/shared/include/protocol_common.h" DESTINATION include)
