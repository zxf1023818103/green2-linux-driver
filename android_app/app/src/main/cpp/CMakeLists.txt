cmake_minimum_required(VERSION 3.22)
project(green2services LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(GREEN2_SRT_INCLUDE "" CACHE PATH "Pinned prebuilt SRT headers")
set(GREEN2_SRT_LIBRARY "" CACHE FILEPATH "Pinned Android libsrtdroid.so")
if(NOT EXISTS "${GREEN2_SRT_INCLUDE}/srt/srt.h" OR NOT EXISTS "${GREEN2_SRT_LIBRARY}")
    message(FATAL_ERROR "Supply the prebuilt SRT headers/library through the Gradle properties")
endif()
add_library(srtdroid SHARED IMPORTED)
set_target_properties(srtdroid PROPERTIES IMPORTED_LOCATION "${GREEN2_SRT_LIBRARY}")
add_library(green2services SHARED native_lib.cpp)
target_include_directories(green2services PRIVATE "${GREEN2_SRT_INCLUDE}")
target_link_libraries(green2services PRIVATE srtdroid android log)
target_link_options(green2services PRIVATE -Wl,--no-undefined)
